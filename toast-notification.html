import { LightningElement, api } from 'lwc';

export default class SfToast extends LightningElement {
    @api title = '';
    @api message = '';
    @api variant = 'info'; // success, error, warning, info
    @api duration = 5000;
    @api isVisible = false;

    get toastClass() {
        return `slds-toast slds-toast_${this.variant}`;
    }

    get isSuccess() {
        return this.variant === 'success';
    }

    get isError() {
        return this.variant === 'error';
    }

    get isWarning() {
        return this.variant === 'warning';
    }

    get isInfo() {
        return this.variant === 'info';
    }

    @api
    show() {
        this.isVisible = true;
        if (this.duration > 0) {
            setTimeout(() => this.closeToast(), this.duration);
        }
    }

    closeToast() {
        this.isVisible = false;
    }
}

Usage Notes

  • Custom toast notification with improved styling
  • Four variants: success, error, warning, info
  • Auto-dismiss with configurable duration
  • Manual close with X button
  • Use @api show() method to display toast
  • Parent component can call show() and set properties
