import { LightningElement, track } from 'lwc';

export default class SfCardCarousel extends LightningElement {
    @track currentIndex = 0;
    
    @track cards = [
        {
            id: '1',
            title: 'Q1 Performance Report',
            description: 'Review the quarterly performance metrics and key achievements.',
            badge: 'Published',
            badgeClass: 'slds-badge slds-badge_success',
            imageStyle: 'background: linear-gradient(135deg, #667eea, #764ba2);'
        },
        {
            id: '2',
            title: 'Product Roadmap',
            description: 'Upcoming features and product roadmap for the next quarter.',
            badge: 'Draft',
            badgeClass: 'slds-badge slds-theme_warning',
            imageStyle: 'background: linear-gradient(135deg, #11998e, #38ef7d);'
        },
        {
            id: '3',
            title: 'Team Updates',
            description: 'New team members, promotions, and organizational changes.',
            badge: 'Published',
            badgeClass: 'slds-badge slds-badge_success',
            imageStyle: 'background: linear-gradient(135deg, #ee0979, #ff6a00);'
        },
        {
            id: '4',
            title: 'Customer Success Stories',
            description: 'Featured customer success stories and case studies.',
            badge: 'Published',
            badgeClass: 'slds-badge slds-badge_success',
            imageStyle: 'background: linear-gradient(135deg, #4facfe, #00f2fe);'
        }
    ];

    previousSlide() {
        this.currentIndex = this.currentIndex === 0 
            ? this.cards.length - 1 
            : this.currentIndex - 1;
    }

    nextSlide() {
        this.currentIndex = (this.currentIndex + 1) % this.cards.length;
    }

    goToSlide(event) {
        const index = parseInt(event.target.dataset.index);
        this.currentIndex = index;
    }
}

Usage Notes

  • Custom card carousel not in Salesforce documentation
  • Perfect for dashboards, news feeds, and featured content
  • Previous/Next navigation buttons
  • Dot indicators for slide navigation
  • Supports dynamic card data from JavaScript
  • Fully responsive with horizontal scroll on mobile
