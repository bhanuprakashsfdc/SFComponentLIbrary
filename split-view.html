import { LightningElement, track } from 'lwc';

export default class SfSplitView extends LightningElement {
    @track selectedId = '1';
    @track listTitle = 'Accounts';
    
    @track items = [
        { id: '1', title: 'Acme Corporation', meta: 'San Francisco, CA' },
        { id: '2', title: 'Global Tech Inc', meta: 'New York, NY' },
        { id: '3', title: 'Innovation Labs', meta: 'Austin, TX' },
        { id: '4', title: 'Tech Solutions', meta: 'Seattle, WA' },
        { id: '5', title: 'Digital Dynamics', meta: 'Boston, MA' }
    ];

    get selectedItem() {
        return this.items.find(item => item.id === this.selectedId);
    }

    selectItem(event) {
        const itemId = event.currentTarget.dataset.id;
        this.selectedId = itemId;
    }

    editItem() {
        this.dispatchEvent(new CustomEvent('edit', {
            detail: { id: this.selectedId }
        }));
    }

    deleteItem() {
        this.dispatchEvent(new CustomEvent('delete', {
            detail: { id: this.selectedId }
        }));
    }
}
