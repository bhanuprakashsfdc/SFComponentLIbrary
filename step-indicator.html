Step Indicator

import { LightningElement, track } from 'lwc';

export default class SfStepIndicator extends LightningElement {
    @track steps = [
        {
            id: '1',
            number: 1,
            title: 'Account Details',
            meta: 'Basic information',
            completed: true,
            current: false,
            pending: false,
            hasConnector: true
        },
        {
            id: '2',
            number: 2,
            title: 'Contact Info',
            meta: 'Primary contact',
            completed: true,
            current: false,
            pending: false,
            hasConnector: true
        },
        {
            id: '3',
            number: 3,
            title: 'Review',
            meta: 'Confirm details',
            completed: false,
            current: true,
            pending: false,
            hasConnector: true
        },
        {
            id: '4',
            number: 4,
            title: 'Submit',
            meta: 'Complete form',
            completed: false,
            current: false,
            pending: true,
            hasConnector: false
        }
    ];

    goToStep(event) {
        const stepId = event.currentTarget.dataset.id;
        this.steps = this.steps.map(step => ({
            ...step,
            completed: step.id < stepId,
            current: step.id === stepId,
            pending: step.id > stepId
        }));
    }
}

Usage Notes

  • Custom step indicator not available in standard Salesforce documentation
  • Perfect for multi-step forms, wizards, and onboarding flows
  • Three states: completed (with checkmark), current (highlighted), pending (grayed)
  • Includes connector lines between steps
  • Each step can have a title and meta description
  • Fully responsive and LWC compatible
